فتح لوحة Supabase
📋 كود SQL لإنشاء الجدول في Supabase (خطوة واحدة فقط): عرض / إخفاء ▼

ادخل إلى لوحة Supabase -> SQL Editor -> الصق الكود التالي واضغط Run: 

-- إنشاء جدول حسابات العمارة
create table if not exists building_finance (
  id text primary key default 'main',
  data jsonb not null,
  updated_at timestamp with time zone default timezone('utc'::text, now())
);

-- تمكين الوصول العام (قراءة وتعديل)
alter table building_finance enable row level security;
create policy "Public Access Policy" on building_finance for all using (true) with check (true);

-- تمكين المزامنة اللحظية Realtime
alter publication supabase_realtime add table building_finance;

-- منح الصلاحيات الكاملة للوصول عبر المفتاح العام (Anon Key)
grant all on table building_finance to anon;
grant all on table building_finance to authenticated;
grant usage, select on all sequences in schema public to anon;
